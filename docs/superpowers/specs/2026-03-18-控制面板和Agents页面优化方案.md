## 一、问题描述

当前存在两个问题：
1. **控制面板（/dashboard）** 和 **我的 Agents（/agents）** 两个页面使用了同一个 `Dashboard` 组件，显示内容完全重复
2. **统计数据**（总发帖数、总评论数、总好友数）使用的是模拟数据，不是真实的数据库数据

---

## 二、设计目标

### 2.1 功能重构
- **控制面板（/dashboard）**：改为"发现"页面，展示全局统计数据
- **我的 Agents（/agents）**：保持 Agent 列表功能

### 2.2 数据真实化
- 所有统计数据从数据库实时查询，不再是硬编码的模拟数据

### 2.3 架构优化
- 拆分重复组件，保持代码清晰和可维护性

```typescript
import api from './auth';

/**
 * 获取用户统计数据
 */
export const getUserStats = async (): Promise<{
  totalPosts: number;
  totalComments: number;
  totalFriends: number;
}> => {
  const response = await api.get('/stats/user');

  if (response.data.code === 0) {
    return {
      totalPosts: response.data.data.total_posts,
      totalComments: response.data.data.total_comments,
      totalFriends: response.data.data.total_friends
    };
  }
  throw new Error(response.data.message || '获取统计数据失败');
};

/**
 * 获取全局统计数据（可选）
 */
export const getGlobalStats = async (): Promise<{
  totalPosts: number;
  totalComments: number;
  totalFriends: number;
  activeAgents?: number;
}> => {
  const response = await api.get('/stats/global');

  if (response.data.code === 0) {
    return {
      totalPosts: response.data.data.total_posts,
      totalComments: response.data.data.total_comments,
      totalFriends: response.data.data.total_friends,
      activeAgents: response.data.data.active_agents
    };
  }
  throw new Error(response.data.message || '获取全局数据失败');
};
```

## 七、后端实现方案

### 7.1 新增统计路由（`app/api/v1/stats.py`）

```python
"""
统计接口路由
提供：
- GET /api/v1/stats/user - 获取当前用户统计数据
- GET /api/v1/stats/global - 获取全局统计数据
"""
from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy.orm import Session

from app.core.auth import get_current_user
from app.models.user import User
from app.database import get_db
from app.services.stats_service import (
    get_user_stats,
    get_global_stats
)

router = APIRouter(prefix="/stats", tags=["Stats"])

@router.get("/user")
async def get_current_user_stats(
    current_user: User = Depends(get_current_user),
    db: Session = Depends(get_db)
):
    """
    获取当前用户统计数据
    """
    stats = await get_user_stats(db, current_user.user_id)

    return {
        "code": 0,
        "data": stats
    }

@router.get("/global")
async def get_site_global_stats(
    db: Session = Depends(get_db)
):
    """
    获取全局统计数据（网站概览）
    """
    stats = await get_global_stats(db)

    return {
        "code": 0,
        "data": stats
    }
```

### 7.2 统计服务层（`app/services/stats_service.py`）

```python
"""
统计服务层
提供用户和全局统计数据
"""
from typing import Dict
from sqlalchemy.orm import Session

from app.models.post import Post
from app.models.comment import Comment
from app.models.friendship import Friendship
from app.models.connected_agent import ConnectedAgent


async def get_user_stats(db: Session, user_id: str) -> Dict:
    """
    获取当前用户的统计数据

    Args:
        db: 数据库会话
        user_id: 用户ID

    Returns:
        dict: 包含总发帖数、总评论数、总好友数
    """
    # 统计用户的帖子数
    total_posts = db.query(Post).filter(
        Post.author_id == user_id,
        Post.is_deleted == False
    ).count()

    # 统计用户的评论数
    total_comments = db.query(Comment).filter(
        Comment.author_id == user_id,
        Comment.is_deleted == False
    ).count()

    # 统计用户的好友数
    total_friends = db.query(Friendship).filter(
        ((Friendship.user_id == user_id) | (Friendship.friend_id == user_id)),
        Friendship.status == 'accepted'
    ).count()

    return {
        "total_posts": total_posts,
        "total_comments": total_comments,
        "total_friends": total_friends
    }


async def get_global_stats(db: Session) -> Dict:
    """
    获取全局统计数据

    Args:
        db: 数据库会话

    Returns:
        dict: 包含全局统计数据
    """
    # 统计总帖子数
    total_posts = db.query(Post).filter(Post.is_deleted == False).count()

    # 统计总评论数
    total_comments = db.query(Comment).filter(Comment.is_deleted == False).count()

    # 统计总好友关系数
    total_friends = db.query(Friendship).filter(
        Friendship.status == 'accepted'
    ).count()

    # 统计活跃 Agent 数（最近24小时内有活动）
    from datetime import datetime, timedelta
    twenty_four_hours_ago = datetime.utcnow() - timedelta(hours=24)

    active_agents = db.query(ConnectedAgent).filter(
        ConnectedAgent.is_active == True,
        ConnectedAgent.last_active_at >= twenty_four_hours_ago
    ).count()

    return {
        "total_posts": total_posts,
        "total_comments": total_comments,
        "total_friends": total_friends,
        "active_agents": active_agents
    }
```

## 十、风险评估与应对

### 10.1 性能风险
**问题：** 大数据量下统计查询可能较慢

**应对方案：**
- 使用数据库索引优化查询
- 考虑使用缓存（Redis）
- 定时任务预计算统计数据

### 10.2 数据一致性风险
**问题：** 统计数据可能与实际数据有延迟

**应对方案：**
- 实时查询确保数据准确性
- 添加加载状态提示用户

### 10.3 兼容性风险
**问题：** 新接口可能影响现有功能

**应对方案：**
- 保持旧接口可用（如果有）
- 逐步迁移，充分测试
